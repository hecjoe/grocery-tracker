  <script>
    // STATE MANAGEMENT
    let state = {
      tripName: "Grocery Run",
      isEditingBudgets: false,
      selectedProfileId: null,
      profiles: [
        { id: '1', name: "Me", budget: 300 }
      ],
      items: []
    };

    // LOCAL STORAGE HANDLING
    function loadData() {
      const saved = localStorage.getItem('costco_tracker_data');
      if (saved) {
        state = JSON.parse(saved);
        state.isEditingBudgets = false;
      }
      if (state.profiles.length > 0 && !state.selectedProfileId) {
        state.selectedProfileId = state.profiles[0].id;
      }
      render();
    }

    function saveData() {
      state.tripName = document.getElementById('tripName').value;
      localStorage.setItem('costco_tracker_data', JSON.stringify(state));
      render();
    }

    // CALCULATIONS
    function totalSpent(profileId) {
      return state.items
        .filter(item => item.profileId === profileId)
        .reduce((sum, item) => sum + (item.unitPrice * item.quantity), 0);
    }

    // RENDER FUNCTIONS
    function render() {
      document.getElementById('tripName').value = state.tripName;
      document.getElementById('editBudgetsBtn').innerText = state.isEditingBudgets ? "Done" : "Edit Budgets";

      // Render Dashboard
      const dashboard = document.getElementById('budgetDashboard');
      dashboard.innerHTML = '';

      state.profiles.forEach(profile => {
        const spent = totalSpent(profile.id);
        const remaining = profile.budget - spent;

        const card = document.createElement('div');
        card.className = 'budget-card';

        if (state.isEditingBudgets) {
          card.innerHTML = `
            <input type="text" value="${profile.name}" onchange="updateProfile('${profile.id}', 'name', this.value)">
            <input type="number" value="${profile.budget}" onchange="updateProfile('${profile.id}', 'budget', parseFloat(this.value) || 0)">
            <button onclick="deleteProfile('${profile.id}')" style="color:var(--accent-red); background:none; border:none; text-align:left; font-size:0.75rem; cursor:pointer;">Delete</button>
          `;
        } else {
          card.innerHTML = `
            <span class="budget-name">${profile.name}</span>
            <span class="budget-amount ${remaining >= 0 ? 'amount-green' : 'amount-red'}">$${remaining.toFixed(2)}</span>
          `;
        }
        dashboard.appendChild(card);
      });

      if (state.isEditingBudgets) {
        const addBtn = document.createElement('button');
        addBtn.className = 'btn-add-profile';
        addBtn.innerHTML = `<span>+</span><span style="font-size:0.75rem">Add</span>`;
        addBtn.onclick = addProfile;
        dashboard.appendChild(addBtn);
      }

      // Render Item List
      const list = document.getElementById('itemList');
      list.innerHTML = '';

      state.items.forEach(item => {
        const profile = state.profiles.find(p => p.id === item.profileId);
        const total = item.unitPrice * item.quantity;

        const row = document.createElement('div');
        row.className = 'item-row';
        row.innerHTML = `
          <div class="item-details">
            <span class="item-name">${item.name}</span>
            <span class="item-buyer">${profile ? profile.name : 'Unknown'}</span>
          </div>
          <div class="item-controls">
            <div class="stepper">
              <button onclick="updateQuantity('${item.id}', -1)">-</button>
              <span>${item.quantity}</span>
              <button onclick="updateQuantity('${item.id}', 1)">+</button>
            </div>
            <span style="font-size:0.85rem; color:var(--text-secondary)">x $${item.unitPrice.toFixed(2)}</span>
            <span style="font-weight:bold; min-width:60px; text-align:right">$${total.toFixed(2)}</span>
            <button class="btn-delete-item" onclick="deleteItem('${item.id}')">✕</button>
          </div>
        `;
        list.appendChild(row);
      });

      // Render Shopper Picker
      const picker = document.getElementById('shopperPicker');
      picker.innerHTML = '';

      state.profiles.forEach(profile => {
        const btn = document.createElement('button');
        btn.className = `picker-btn ${state.selectedProfileId === profile.id ? 'active' : ''}`;
        btn.innerText = profile.name;
        btn.onclick = () => {
          state.selectedProfileId = profile.id;
          render();
        };
        picker.appendChild(btn);
      });

      validateInput();
    }

    // USER ACTIONS
    function toggleEditBudgets() {
      state.isEditingBudgets = !state.isEditingBudgets;
      render();
    }

    function addProfile() {
      const id = Date.now().toString();
      state.profiles.push({ id, name: "New", budget: 50 });
      if (!state.selectedProfileId) state.selectedProfileId = id;
      saveData();
    }

    function updateProfile(id, key, value) {
      const profile = state.profiles.find(p => p.id === id);
      if (profile) {
        profile[key] = value;
        saveData();
      }
    }

    function deleteProfile(id) {
      state.profiles = state.profiles.filter(p => p.id !== id);
      if (state.selectedProfileId === id) {
        state.selectedProfileId = state.profiles[0]?.id || null;
      }
      saveData();
    }

    function addItem() {
      const nameInput = document.getElementById('newItemName');
      const priceInput = document.getElementById('newItemPrice');

      const name = nameInput.value.trim();
      const price = parseFloat(priceInput.value);

      if (name && !isNaN(price) && state.selectedProfileId) {
        state.items.push({
          id: Date.now().toString(),
          name,
          unitPrice: price,
          quantity: 1,
          profileId: state.selectedProfileId
        });

        nameInput.value = '';
        priceInput.value = '';
        saveData();
      }
    }

    function updateQuantity(id, delta) {
      const item = state.items.find(i => i.id === id);
      if (item) {
        item.quantity += delta;
        if (item.quantity <= 0) {
          deleteItem(id);
        } else {
          saveData();
        }
      }
    }

    function deleteItem(id) {
      state.items = state.items.filter(i => i.id !== id);
      saveData();
    }

    function clearAllItems() {
      if (confirm("Clear all items from this trip?")) {
        state.items = [];
        saveData();
      }
    }

    function validateInput() {
      const name = document.getElementById('newItemName').value.trim();
      const price = document.getElementById('newItemPrice').value.trim();
      document.getElementById('addItemBtn').disabled = !(name && price && state.selectedProfileId);
    }

    // INITIALIZATION & SERVICE WORKER (OFFLINE SUPPORT)
    loadData();

    if ('serviceWorker' in navigator) {
      const swCode = `
        self.addEventListener('install', e => e.waitUntil(caches.open('pwa-v1').then(c => c.addAll(['./']))));
        self.addEventListener('fetch', e => e.respondWith(caches.match(e.request).then(r => r || fetch(e.request))));
      `;
      const blob = new Blob([swCode], { type: 'application/javascript' });
      navigator.serviceWorker.register(URL.createObjectURL(blob));
    }
  </script>
